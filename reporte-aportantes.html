    // SALIDAS — sin BAJAS SIN FECHA; con peso promedio sacrificio

    var mort = '<div class="kg">'

      + kc('red', num(k.muertes), 'Bajas por muerte', 'del total registrado')

          ? kc('teal', k.pesoPromedioSacrificio + ' kg', 'Peso promedio sacrificio',

               'sobre ' + k.sacrificios + ' animales')

          : '')

      + kc('gray', num(k.ventas), 'Ventas en pie', 'salida comercial, no baja')

      + ((k.sacrificios || k.ventas)

          ? '<div class="crec-nota"><b>Los sacrificios y las ventas no son mortalidad.</b> '

          : '')

      + causas

      + (k.noLocalizados

          ? '<div class="crec-nota"><b>Los no localizados no se cuentan como mortalidad.</b> '

            + (k.noLocalizadosDetalle.length

                ? ' Chapetas: '+k.noLocalizadosDetalle.map(function(a){

                : '')

      + '<div><div style="font-size:10px;font-weight:700;letter-spacing:.8px;text-transform:uppercase;color:#8A9696;margin-bottom:7px">Por raza</div>'

        + 'El hato real a la fecha de corte es <b>'+num(k.hatoActual)+'</b>.</div>'

    // Notas al pie simplificadas

      + '<div><b>Fecha de corte:</b> '+fmtLargo(corte)+'.</div>'

      + '<div style="margin-top:10px;color:#A8B3B3">LAAAMB Corderos · Finca La Marinilla · '

      +   'Aparcería — estos animales no forman parte del inventario de la finca.</div>'

    $('hoja').innerHTML =

        '<div class="rep-head">'

      +   '<div class="rep-head-top">'

      +     '<div><div class="rep-brand">LAAAMB<span>.</span></div>'

  function sec(titulo, html, hint){

    return '<div class="rep-sec"><div class="rep-sec-t"><span>'+esc(titulo)+'</span>'

      + (hint?'<span class="hint">'+esc(hint)+'</span>':'')+'</div>'+html+'</div>';

  }


  function distribucion(grupos, total, etiquetaCol, mapa){

    if(!grupos || !grupos.length){

      return '<div style="font-size:12px;color:#8A9696;padding:10px 0">Sin datos para agrupar.</div>';

    }

    var filas = grupos.map(function(g){

      var p = total ? Math.round((g.n/total)*1000)/10 : 0;

      var nombre = (mapa && mapa[g.clave]) ? mapa[g.clave] : g.clave;

      return '<tr>'

        + '<td>'+esc(nombre)+'</td>'

        + '<td class="num">'+g.n+'</td>'

        + '<td class="num">'+String(p).replace('.',',')+'%</td>'

        + '<td style="width:34%">'

        +   '<div style="height:8px;background:#EAEEEE;border-radius:99px;overflow:hidden">'

        +     '<div style="height:100%;width:'+Math.max(p,1)+'%;background:#1AA6A6"></div>'

        +   '</div>'

        + '</td></tr>';

    }).join('');

    return '<table class="rt"><thead><tr>'

      + '<th>'+esc(etiquetaCol)+'</th><th class="num">Animales</th>'

      + '<th class="num">%</th><th></th></tr></thead><tbody>'

      + filas + '</tbody></table>';

  }


  var MAPA_SEXO = { hembra: 'Hembras', macho: 'Machos' };

  function kc(cls, n, l, s){

    return '<div class="kc '+cls+'"><div class="n">'+n+'</div><div class="l">'+esc(l)+'</div>'

      + (s?'<div class="s">'+esc(s)+'</div>':'')+'</div>';

  }


  async function generar(){

    var id = $('f-aportante').value;

    if(!id){ toast('Selecciona un aportante', true); return; }

    var corte = $('f-corte').value;

    if(!corte){ toast('Indica la fecha de corte', true); return; }


    $('hoja').innerHTML = '<div class="empty-box">Calculando…</div>';

    try{

      var ap = APORTANTES.filter(function(a){ return a.id === id; })[0];

      if(!ap) throw new Error('Aportante no encontrado');


      var rows = await window.APARCERIA.getAnimales(id);

      if(!rows.length){

        $('hoja').innerHTML = '<div class="empty-box">'

          + '<div style="font-size:15px;color:var(--text2)">Sin animales cargados para '

          + esc(ap.nombre) + '.</div>'

          + '<div style="font-size:12.5px;color:var(--text4);margin-top:8px;max-width:460px;margin-left:auto;margin-right:auto;line-height:1.6">'

          + 'Carga los animales desde <b>Aparcería → Animales</b> y vuelve.</div></div>';

        ACTUAL = null;

        return;

      }

      if(ap.hato_inicial == null){

        toast('Aviso: este aportante no tiene hato_inicial definido — crecimiento y cumplimiento saldrán vacíos', true);

      }


      ACTUAL = { aportante: ap, rows: rows, kpis: window.APARCERIA.kpis(rows, ap), corte: corte };

      render();

      toast('Reporte generado');

    }catch(e){

      var m = fallo('Generar reporte', e);

      $('hoja').innerHTML = '<div class="empty-box" style="color:var(--red)">'+esc(m)+'</div>';

      ACTUAL = null;

    }

  }


  function exportarExcel(){

    if(!ACTUAL){ toast('Genera el reporte primero', true); return; }

    try{

      if(!window.XLSX) throw new Error('SheetJS no cargó (revisa la conexión)');

      var k=ACTUAL.kpis, ap=ACTUAL.aportante;

      var kpiRows = [

        ['REPORTE DE APARCERÍA — LAAAMB Corderos'],

        ['Aportante', ap.nombre],

        ['Fecha de corte', ACTUAL.corte],

        [],

        ['CRECIMIENTO DEL HATO'],

        ['Madres aportadas al inicio', k.hatoInicial],

        ['Nacidas en el proyecto (vivas)', k.criasActivas],

        ['Hato actual', k.hatoActual],

        ['Crecimiento %', k.crecimientoPct],

        ['Crecimiento neto (animales)', k.crecimientoNeto],

        ['Meta anual', k.meta],

        ['Cumplimiento meta %', k.cumplimientoPct],

        [],

        ['COMPOSICIÓN'],

        ['Total registrados (reales)', k.totalRegistradas],

        ['Fundadoras activas', k.fundadorasActivas],

        ['Crias activas', k.criasActivas],

        ['Corderas en levante', k.corderasLevante],

        ['Prolificidad', k.prolificidad],

        [],

        ['SALIDAS'],

        ['Bajas por muerte', k.muertes],

        ['% Mortalidad', k.mortalidadPct],

        ['Sacrificios', k.sacrificios],

        ['Peso promedio sacrificio (kg)', k.pesoPromedioSacrificio],

        ['Ventas en pie', k.ventas],

        ['No localizados', k.noLocalizados]

      ];

      if(k.causasBaja.length){

        kpiRows.push([], ['CAUSAS DE MORTALIDAD'], ['Causa','Animales','%']);

        k.causasBaja.forEach(function(c){

          var p = k.muertes ? Math.round((c.n/k.muertes)*1000)/10 : 0;

          kpiRows.push([c.clave, c.n, p]);

        });

      }


      var det = [['Chapeta','Codigo interno','Origen','Raza','Sexo','Tipo','Grupo','Estado salida',

                  'Fecha salida','Motivo salida','Peso salida','Estado reproductivo',

                  'Fecha nacimiento','Madre','Peso vigente kg','Tipo peso','Fecha peso','Notas']];

      ACTUAL.rows.forEach(function(a){

        var p = window.APARCERIA.pesoVigente(a);

        det.push([a.codigo_original||a.codigo, a.codigo, a.origen, a.raza, a.sexo, a.tipo, a.grupo, a.estado_salida,

                  a.fecha_salida, a.motivo_salida, a.peso_salida,

                  a.estado_reproductivo, a.fecha_nacimiento, a.madre_codigo,

                  p?p.peso_kg:null, p?p.tipo:null, p?p.fecha:null, a.notas]);

      });


      var wb = window.XLSX.utils.book_new();

      window.XLSX.utils.book_append_sheet(wb, window.XLSX.utils.aoa_to_sheet(kpiRows), 'KPIs');

      window.XLSX.utils.book_append_sheet(wb, window.XLSX.utils.aoa_to_sheet(det), 'Animales');

      var slug = ap.nombre.replace(/[^A-Za-z0-9]+/g,'_').replace(/^_|_$/g,'');

      window.XLSX.writeFile(wb, 'Reporte_Aparceria_'+slug+'_'+ACTUAL.corte+'.xlsx');

      toast('Excel exportado');

    }catch(e){ fallo('Exportar Excel', e); }

  }


  function imprimir(){

    if(!ACTUAL){ toast('Genera el reporte primero', true); return; }

    window.print();

  }


  async function init(){

    try{

      APORTANTES = await window.APARCERIA.getAportantes();

      var sel = $('f-aportante');

      if(!APORTANTES.length){

        sel.innerHTML = '<option value="">Sin aportantes</option>';

        $('hoja').innerHTML = '<div class="empty-box">No hay aportantes registrados.</div>';

        return;

      }

      sel.innerHTML = APORTANTES.map(function(a){

        return '<option value="'+esc(a.id)+'">'+esc(a.nombre)+'</option>';

      }).join('');

    }catch(e){

      fallo('Cargar aportantes', e);

      $('hoja').innerHTML = '<div class="empty-box" style="color:var(--red)">'+esc(e.message)+'</div>';

    }

  }


  function whenReady(cb){

    if(window.AUTH_ROL){ cb(); return; }

    var n=0;

    var iv=setInterval(function(){ if(window.AUTH_ROL || n>50){ clearInterval(iv); cb(); } n++; }, 100);

  }


  document.addEventListener('DOMContentLoaded', function(){

    $('f-corte').value = new Date().toISOString().split('T')[0];

    $('btn-generar').addEventListener('click', function(){ generar(); });

    $('btn-pdf').addEventListener('click', imprimir);

    $('btn-excel').addEventListener('click', exportarExcel);

    whenReady(init);

  });

})();
